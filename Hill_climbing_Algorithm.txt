N = 3

class PuzzleBoard:
    def __init__(self, board, zero_x, zero_y, heuristic):
        self.board = board
        self.zero_x = zero_x
        self.zero_y = zero_y
        self.heuristic = heuristic

dx = [0, 0, -1, 1]
dy = [-1, 1, 0, 0]

def is_valid(x, y):
    return 0 <= x < N and 0 <= y < N

def is_goal(board):
    goal = [
        [1, 2, 3],
        [4, 5, 6],
        [7, 8, 0]
    ]
    return board == goal

def manhattan_distance(board):
    goal_pos = {
        1: (0, 0), 2: (0, 1), 3: (0, 2),
        4: (1, 0), 5: (1, 1), 6: (1, 2),
        7: (2, 0), 8: (2, 1)
    }
    distance = 0
    for i in range(N):
        for j in range(N):
            val = board[i][j]
            if val != 0:
                gx, gy = goal_pos[val]
                distance += abs(i - gx) + abs(j - gy)
    return distance

def hill_climbing(start_board, zero_x, zero_y):
    current = PuzzleBoard(start_board, zero_x, zero_y, manhattan_distance(start_board))
    while True:
        if is_goal(current.board):
            print("Goal Found using Hill Climbing")
            for row in current.board:
                print(row)
            return
        neighbors = []
        for i in range(4):
            nx = current.zero_x + dx[i]
            ny = current.zero_y + dy[i]
            if is_valid(nx, ny):
                new_board = [r[:] for r in current.board]
                new_board[current.zero_x][current.zero_y], new_board[nx][ny] = new_board[nx][ny], new_board[current.zero_x][current.zero_y]
                h = manhattan_distance(new_board)
                neighbors.append(PuzzleBoard(new_board, nx, ny, h))
        best_neighbor = min(neighbors, key=lambda s: s.heuristic)
        if best_neighbor.heuristic >= current.heuristic:
            print("Stuck at Local Maximum")
            return
        current = best_neighbor

start_board = [
    [1, 2, 3],
    [4, 5, 6],
    [7, 0, 8]
]

hill_climbing(start_board, 2, 1)
