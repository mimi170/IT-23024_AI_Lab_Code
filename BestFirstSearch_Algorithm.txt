
import heapq

goal = [
    [1, 2, 3],
    [4, 5, 6],
    [7, 8, 0]
]

row = [0, 0, -1, 1]
col = [-1, 1, 0, 0]

def heuristic(board, goal):
    position = {}
    for i in range(3):
        for j in range(3):
            position[goal[i][j]] = (i, j)

    h = 0
    for i in range(3):
        for j in range(3):
            tile = board[i][j]
            if tile != 0:
                gx, gy = position[tile]
                h += abs(i - gx) + abs(j - gy)
    return h

def find_zero(board):
    for i in range(3):
        for j in range(3):
            if board[i][j] == 0:
                return i, j

def to_tuple(board):
    return tuple(tuple(r) for r in board)

def best_first(start):
    pq = []
    visited = set()

    h = heuristic(start, goal)
    heapq.heappush(pq, (h, start))

    while pq:
        h, board = heapq.heappop(pq)

        if board == goal:
            print("Goal Reached!")
            print("Heuristic =", h)
            print("Answer Board:")
            for r in board:
                print(r)
            return

        visited.add(to_tuple(board))
        x, y = find_zero(board)

        for i in range(4):
            nx = x + row[i]
            ny = y + col[i]

            if 0 <= nx < 3 and 0 <= ny < 3:
                new_board = [r[:] for r in board]
                new_board[x][y], new_board[nx][ny] = \
                    new_board[nx][ny], new_board[x][y]

                if to_tuple(new_board) not in visited:
                    new_h = heuristic(new_board, goal)
                    heapq.heappush(pq, (new_h, new_board))

start = [
    [1, 2, 3],
    [4, 6, 0],
    [7, 5, 8]
]

best_first(start)
